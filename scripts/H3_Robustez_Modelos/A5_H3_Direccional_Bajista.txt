def _ssr_conc_a5(y, rm, s, gamma_t, c):
    G = _G_log(s, gamma_t, c)
    X = np.column_stack([np.ones(len(y)), G, rm, G * rm])
    coef = np.linalg.lstsq(X, y, rcond=None)[0]
    u = y - X @ coef
    return float(u @ u)

def kappa_a5(a, b):
    a, b = np.asarray(a), np.asarray(b)
    po = (a == b).mean()
    pe = (a == 1).mean() * (b == 1).mean() + (a == 0).mean() * (b == 0).mean()
    return (po - pe) / (1 - pe) if pe < 1 else np.nan

def _capm_inter(y, rm, w):
    """CAPM condicional con interacción w (dummy o G continua)."""
    if np.ptp(w) == 0:
        return None
    X = sm.add_constant(np.column_stack([w, rm, w * rm]))
    m = sm.OLS(y, X).fit(cov_type='HAC', cov_kwds={'maxlags': 3})
    return {'delta': m.params[3], 'se': m.bse[3],
            't': m.tvalues[3], 'p': m.pvalues[3]}


def _decision(r):
    if r is None:
        return 'n/d'
    if r['p'] < 0.05:
        return 'refugio' if r['delta'] < 0 else 'amplifica'
    return 'sin cambio'

def _delta_r(y, rm, w):
    if np.ptp(w) == 0:
        return np.nan
    X = np.column_stack([np.ones(len(y)), w, rm, w * rm])
    return np.linalg.lstsq(X, y, rcond=None)[0][3]


def _idx_bloques_a5(n, largo, rng):
    n_b = int(np.ceil(n / largo))
    starts = rng.integers(0, n - largo + 1, size=n_b)
    return np.concatenate([np.arange(s, s + largo) for s in starts])[:n]
