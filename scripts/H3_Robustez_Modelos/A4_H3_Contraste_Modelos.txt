def _capm_dummy(y, rm, d, hac=True):
    """CAPM condicional; devuelve (δ, se, p, ssr) o NaN si d no varía."""
    if d.sum() == 0 or d.sum() == len(d):
        return np.nan, np.nan, np.nan, np.nan
    X = sm.add_constant(np.column_stack([d, rm, d * rm]))
    if hac:
        m = sm.OLS(y, X).fit(cov_type='HAC', cov_kwds={'maxlags': 3})
    else:
        m = sm.OLS(y, X).fit()
    return m.params[3], m.bse[3], m.pvalues[3], m.ssr

def kappa_cohen(a, b):
    a, b = np.asarray(a), np.asarray(b)
    po = (a == b).mean()
    pe = (a == 1).mean() * (b == 1).mean() + (a == 0).mean() * (b == 0).mean()
    return (po - pe) / (1 - pe) if pe < 1 else np.nan

def _delta_rapido(y, rm, d):
    if d.sum() == 0 or d.sum() == len(d):
        return np.nan
    X = np.column_stack([np.ones(len(y)), d, rm, d * rm])
    return np.linalg.lstsq(X, y, rcond=None)[0][3]


def _idx_bloques(n, largo, rng):
    n_b = int(np.ceil(n / largo))
    starts = rng.integers(0, n - largo + 1, size=n_b)
    return np.concatenate([np.arange(s, s + largo) for s in starts])[:n]


boot_rows = []
for score in SCORES_A4:
    for cartera in CARTERAS_A4:
        d = datos_cache[(score, cartera)]
        y, rm = d['Y'].values, d['Rm_Rf'].values
        dms = d['D_MS'].values.astype(float)
        dg  = d['D_G'].values.astype(float)
        dl  = d['D_L'].values.astype(float)
        n = len(y)
        dif_g, dif_l = [], []
        for _ in range(B_BOOT):
            ii = _idx_bloques(n, BLOQUE, rng)
            d_ms_b = _delta_rapido(y[ii], rm[ii], dms[ii])
            d_g_b  = _delta_rapido(y[ii], rm[ii], dg[ii])
            d_l_b  = _delta_rapido(y[ii], rm[ii], dl[ii])
            if not np.isnan(d_ms_b):
                if not np.isnan(d_g_b): dif_g.append(d_g_b - d_ms_b)
                if not np.isnan(d_l_b): dif_l.append(d_l_b - d_ms_b)
        for et, difs in [('GARCH−MS', np.array(dif_g)), ('LSTARD−MS', np.array(dif_l))]:
            lo, hi = np.percentile(difs, [2.5, 97.5])
            p_boot = 2 * min((difs <= 0).mean(), (difs >= 0).mean())
            boot_rows.append({'score': score, 'cartera': cartera, 'dif': et,
                              'media': difs.mean(), 'ic_lo': lo, 'ic_hi': hi,
                              'p_boot': p_boot, 'contiene_0': lo <= 0 <= hi})

        ics = {}
        for m, col in [('MS', 'D_MS'), ('GARCH', 'D_G')]:
            _, _, _, ssr = _capm_dummy(y, rm, d[col].values.astype(float), hac=False)
            ics[f'AIC_{m}'] = n * np.log(ssr / n) + 2 * 4
            ics[f'BIC_{m}'] = n * np.log(ssr / n) + 4 * np.log(n)
        G = d['G_L'].values
        X = sm.add_constant(np.column_stack([G, rm, G * rm]))
        ssr_l = sm.OLS(y, X).fit().ssr
        ics['AIC_LSTAR'] = n * np.log(ssr_l / n) + 2 * 6
        ics['BIC_LSTAR'] = n * np.log(ssr_l / n) + 6 * np.log(n)
        mejor = min(['MS', 'GARCH', 'LSTAR'], key=lambda m: ics[f'AIC_{m}'])
        aic_rows.append({'score': score, 'cartera': cartera, **ics, 'mejor': mejor})
        print(f"  {LABELS_A4[score]:<15} {cartera:<7} {ics['AIC_MS']:>8.1f} "
              f"{ics['AIC_GARCH']:>10.1f} {ics['AIC_LSTAR']:>10.1f} {ics['BIC_MS']:>8.1f} "
              f"{ics['BIC_GARCH']:>10.1f} {ics['BIC_LSTAR']:>10.1f}  {mejor}")
