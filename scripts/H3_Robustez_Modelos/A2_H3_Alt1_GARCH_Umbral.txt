# MOTIVACIÓN (observación del tribunal, Hito 4): el MS(2) clasifica solo
# 18 de 120 meses como crisis (15%) — regímenes DESBALANCEADOS. El δ del
# CAPM condicional se estima con pocas observaciones en el régimen de crisis.
#
# ALTERNATIVA 1 — GARCH(1,1)-t con umbral de cuantil:
#   σ²_t = ω + α·ε²_{t-1} + β·σ²_{t-1}   (retornos diarios del mercado)
#   Mes en CRISIS si la volatilidad condicional media del mes supera el
#   cuantil P75 de la distribución mensual → 25% de meses en crisis
#   POR CONSTRUCCIÓN. El balance deja de ser un resultado endógeno de una
#   mezcla gaussiana y pasa a ser una decisión de diseño transparente,
#   con sensibilidad P70/P80/P85 (Script A4).
#
# POR QUÉ RESPONDE A LA CRÍTICA:
#   (a) No estima mezclas → no hay riesgo de régimen degenerado ni de
#       clasificación dominada por un solo episodio.
#   (b) El grado de balance es explícito y contrastable (análisis de
#       sensibilidad del umbral en A4).
#   (c) GARCH(1,1) es el modelo canónico de volatilidad (Bollerslev 1986);
#       la clasificación por umbral de volatilidad sigue a Schwert (1989).
#
# ETAPA 2: idéntica al Script 3.4 (CAPM condicional OLS + Newey-West),
# solo cambia la dummy de régimen → los δ son directamente comparables.
#
# REQUIERE EN MEMORIA (ejecutar ANTES el pipeline completo hasta Script 3.4):
#   rm_diario (3.4) + ports_raw (3.2) + df_factors_4f (3.1)

UMBRAL_GARCH_Q = 0.75   # cuantil principal (crisis = cuartil superior de vol.)


def _stars_a2(p):
    if   p < 0.01: return '***'
    elif p < 0.05: return '**'
    elif p < 0.10: return '*'
    else:          return ''

# ==============================================================================
# SECCIÓN 1: GARCH(1,1)-t SOBRE RETORNOS DIARIOS DEL MERCADO
# ==============================================================================

print("─" * 80)
print("SECCIÓN 1: GARCH(1,1)-t sobre el retorno diario del STOXX 600")
print("─" * 80)

# arch trabaja mejor con retornos en % (estabilidad numérica)
ret_pct = rm_diario * 100

mod_g = arch_model(ret_pct, mean='Constant', vol='GARCH', p=1, q=1, dist='t')
res_garch_fit = mod_g.fit(disp='off')

omega_g = res_garch_fit.params['omega']
alpha_g = res_garch_fit.params['alpha[1]']
beta_g  = res_garch_fit.params['beta[1]']
nu_g    = res_garch_fit.params.get('nu', np.nan)
persist = alpha_g + beta_g
half_life = np.log(0.5) / np.log(persist) if persist < 1 else np.inf

# Volatilidad condicional diaria (%/día) y agregación mensual
sigma_d_garch = pd.Series(res_garch_fit.conditional_volatility.values,
                          index=ret_pct.index, name='sigma_garch')
sigma_m_garch = sigma_d_garch.groupby(sigma_d_garch.index.to_period('M')).mean()
fechas_m_g = sigma_m_garch.index.to_timestamp()
sigma_m_garch = pd.Series(sigma_m_garch.values, index=fechas_m_g, name='sigma_m')

print(f"""
  PARÁMETROS GARCH(1,1)-t ({len(ret_pct):,} días):
    ω = {omega_g:.5f} | α = {alpha_g:.4f} | β = {beta_g:.4f} | ν (g.l. t) = {nu_g:.2f}
    Persistencia α+β = {persist:.4f} | Semivida del shock = {half_life:.1f} días
    Log-verosimilitud = {res_garch_fit.loglikelihood:.1f} | AIC = {res_garch_fit.aic:.1f} | BIC = {res_garch_fit.bic:.1f}

  Volatilidad condicional mensual (media de σ_t diaria, %/día):
    mín = {sigma_m_garch.min():.3f} | mediana = {sigma_m_garch.median():.3f} | máx = {sigma_m_garch.max():.3f}
    (anualizada: mín {sigma_m_garch.min()*np.sqrt(252):.1f}% | máx {sigma_m_garch.max()*np.sqrt(252):.1f}%)
""")

df_regimen_garch = pd.DataFrame({
    'Date': D_crisis_garch.index,

def capm_condicional_garch(score, cartera):
    df_p = ports_raw[score].merge(
        df_factors_4f[['YearMonth', 'Rm_Rf']], on='YearMonth', how='inner')
    df_p['Date'] = pd.to_datetime(df_p['Date'])
    df_p = df_p.merge(df_regimen_garch, on='Date', how='inner')

    ret_col = 'EX_LS' if cartera == 'Q1-Q5' else f'EX_{cartera}'
    datos = df_p[['Date', ret_col, 'Rm_Rf', 'D_crisis_g']].dropna().sort_values('Date')
    if len(datos) < 24:
        return None

    datos['D_x_Rm'] = datos['D_crisis_g'] * datos['Rm_Rf']
    Y = datos[ret_col].values
    X = sm.add_constant(datos[['D_crisis_g', 'Rm_Rf', 'D_x_Rm']].values)

    m_nw = sm.OLS(Y, X).fit(cov_type='HAC', cov_kwds={'maxlags': 3})
    m_ols = sm.OLS(Y, X).fit()
    alpha, gamma, beta, delta = m_nw.params

    return {'score': score, 'cartera': cartera, 'n': len(datos),
            'beta_calma': beta, 'beta_crisis': beta + delta,
            'delta': delta, 't_delta': m_nw.tvalues[3], 'p_delta': m_nw.pvalues[3],
            'se_delta': m_nw.bse[3], 'wald': m_nw.tvalues[3]**2,
            'R2': m_ols.rsquared, 'ssr': m_ols.ssr,
            'modelo_nw': m_nw, 'datos': datos}


resultados_h3_garch = {}
for score in SCORES_A2:
    for cartera in CARTERAS_A2:
        r = capm_condicional_garch(score, cartera)
        if r is not None:
            resultados_h3_garch[(score, cartera)] = r
